let loadStartTime = 0;

function startLoading(){

  const bar = document.getElementById("loaderBar");

  loadStartTime = Date.now(); // start time save

  bar.style.width = "30%";

  setTimeout(()=>{ bar.style.width = "60%"; },150);

  setTimeout(()=>{ bar.style.width = "85%"; },400);

}

function finishLoading(){

  const bar = document.getElementById("loaderBar");

  let elapsed = Date.now() - loadStartTime;

  let remaining = 600 - elapsed; // minimum 600ms dikhe

  if(remaining < 0) remaining = 0;

  setTimeout(()=>{

    bar.style.width = "100%";

    setTimeout(()=>{

      bar.style.width = "0%";

    },200);

  }, remaining);

}

function saveHistory(url){

  let history = JSON.parse(localStorage.getItem("history")) || [];

  history.unshift({

    url: url,

    time: Date.now()

  });

  if(history.length > 50){

    history.pop();

  }

  localStorage.setItem("history", JSON.stringify(history));

}
    
function openHistory(){

  let panel = document.getElementById("historyPanel");

  let list = document.getElementById("historyList");

  let history = JSON.parse(localStorage.getItem("history")) || [];

  list.innerHTML = "";

  if(history.length === 0){

    list.innerHTML = "<p style='padding:15px'>No history yet</p>";

    panel.style.display = "flex";

    return;

  }

  let lastDate = "";

  history.forEach(function(item, index){

    let date = new Date(item.time).toDateString();

    if(date !== lastDate){

      let dateDiv = document.createElement("div");

      dateDiv.className = "history-date";

      dateDiv.innerText = date;

      list.appendChild(dateDiv);

      lastDate = date;

    }

    let div = document.createElement("div");

    div.className = "history-item";

    div.innerHTML =

      '<div class="history-left">' +

        '<div class="history-icon">' +

  '<img src="https://www.google.com/s2/favicons?domain=' + item.url + '&sz=64">' +

'</div>' +

        '<div class="history-text">' +

          '<div class="history-url">' + item.url + '</div>' +

          '<div class="history-time">' + new Date(item.time).toLocaleString() + '</div>' +

        '</div>' +

      '</div>' +

      '<span class="material-icons delete-btn" onclick="deleteHistory(event,'+index+')">close</span>';

    div.onclick = function(){

      window.location = item.url;

    };

    list.appendChild(div);

  });

  panel.style.display = "flex";
  
  window.history.pushState({page:"history"}, "", "");

}
    
function closeHistory(){

  document.getElementById("historyPanel").style.display = "none";

}

function deleteHistory(e, index){

  e.stopPropagation(); // 🔥 sabse important

  let history = JSON.parse(localStorage.getItem("history")) || [];

  history.splice(index,1);

  localStorage.setItem("history", JSON.stringify(history));

  openHistory(); // refresh

}

function filterHistory(query){

  let items = document.querySelectorAll(".history-item");

  items.forEach(item=>{

    if(item.innerText.toLowerCase().includes(query.toLowerCase())){

      item.style.display = "flex";

    } else {

      item.style.display = "none";

    }

  });

}
