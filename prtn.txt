function searchWeb(){  
  
var q=document.getElementById("search").value;  
  
window.location="https://www.google.com/search?q="+q;  
  
}  
  
function openSite(url){

window.open(url,"_blank");

} 
  
function startVoice(){

const mic = document.getElementById("micBtn");

mic.classList.add("mic-active");

const recognition = new webkitSpeechRecognition();

recognition.lang="en-IN";

recognition.start();

recognition.onresult=function(event){

let text=event.results[0][0].transcript;

document.getElementById("search").value=text;

mic.classList.remove("mic-active");

window.location="https://www.google.com/search?q="+text;

}

recognition.onend=function(){

mic.classList.remove("mic-active");

}

}  
  
/* ENTER SEARCH */  
  
document.getElementById("search").addEventListener("keypress",function(e){  
  
if(e.key==="Enter"){  
searchWeb();  
}  
  
});  
  
/* GOOGLE SUGGESTIONS */  
  
const searchInput=document.getElementById("search");  
  
searchInput.addEventListener("keyup",function(){  
  
let query=this.value;  
  
if(query.length<2){  
  
document.getElementById("suggestions").innerHTML="";  
  
return;  
  
}  
  
let script=document.createElement("script");  
  
script.src="https://suggestqueries.google.com/complete/search?client=chrome&q="+query+"&callback=showSuggestions";  
  
document.body.appendChild(script);  
  
});  
  
function showSuggestions(data){  
  
let box=document.getElementById("suggestions");  
  
box.innerHTML="";  
  
data[1].forEach(function(item){  
  
let div=document.createElement("div");  
  
div.className="suggestion-item";  
  
div.textContent=item;  
  
div.onclick=function(){  
  
document.getElementById("search").value=item;  
  
box.innerHTML="";  
  
searchWeb();  
  
};  
  
box.appendChild(div);  
  
});  
  
}

// close apps panel when tapping outside

document.addEventListener("click", function(e){

let panel = document.getElementById("appsPanel");

let appBtn = document.querySelector(".navicon[onclick*='openApps']");

if(panel.classList.contains("active") && 

   !panel.contains(e.target) && 

   !appBtn.contains(e.target)){

       

   panel.classList.remove("active");

}

});

  panel.style.transition = "0.3s ease";

    panel.classList.remove("active");

    panel.style.transform = "translateY(0)";

  },300);

}
